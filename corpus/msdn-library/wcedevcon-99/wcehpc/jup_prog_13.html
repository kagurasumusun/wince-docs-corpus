Using the TNEF API

Windows CE 2.10 contains several encoding and decoding options, including the ability to package and parse meeting request information to and from the Transport-Neutral Encapsulation Format (TNEF).TNEF is a Mail API –defined format that encapsulates MAPI message properties inside a single binary stream. To preserve MAPI properties, a sending application uses TnefBuildInfo to encode a MAPI message into a TNEF stream. A receiving application uses TnefGetInfo to decode the stream and reconstruct the original MAPI message. Currently, the transport DLL that ships with allows applications to retrieve attachments and meeting-request properties.

TnefEncodeBuffer and TnefEncodeFile functions take binary data and translate it into 7-bit data. A server can then send this information over the Internet, and the receiving application can decode this information, using TnefDecodeBuffer and TnefDecodeFile. These functions translate to and from Base64, Quoted-Printable, and UUENCODE formats. Once you decode the buffer or file, you can translate the data into Unicode using TnefConvertToUnicode. To translate directly between Modified UTF7 format and Unicode, use ModifiedUTF7ToUnicode and UnicodeToModifiedUTF7.

Note These functions begin with the letters TNEF because they appear in the TNEF API; they are not directly related to the TNEF format.

The following code example shows how to use TnefEncodeFile.

BOOL SendAttFile (HANDLE hMail, MailMsg mm)
{
  MailAtt mAtt;                       // The MailAtt structure
  
  // Check if there are any attachments to the mail.
  if (!(mm.dwFlags & MAIL_STATUS_ATTACHMENTS))
    return FALSE;

  mAtt.uiAttachmentNumber = 1;

  while (MailGetAttachment(hMail, &mm, &mAtt) != -1 )
  {
    if (mAtt.szLocalName)
    {
      HANDLE hFile;                   // Handle to the encoded file
      LPTSTR lpszEncodedFile = NULL;  // Pointer to the encoded file 
                                      //name
      
      // Encodes the attachment file
      if (!(lpszEncodedFile = TnefEncodeFile (
                                              mAtt.szLocalName, 
                                              CODE_B64)))
      {
        return FALSE;
      }

      // Open the encoded file for reading.
      if ((hFile = CreateFile (
                              lpszEncodedFile, 
                              GENERIC_READ, 
                              FILE_SHARE_READ, 
                              (LPSECURITY_ATTRIBUTES)0, 
                              OPEN_EXISTING, 
                              FILE_ATTRIBUTE_NORMAL, 
                              (HANDLE)0)) != INVALID_HANDLE_VALUE)
      {
        // Insert code here for reading data from the encoded file and
        // then sending the data out.
        // ...
// Once the reading and sending data are done, close the file.
        CloseHandle(hFile);
      }
      
      // Delete the encoded file, and free the resource associated with 
      // lpszEncodedFile.
      DeleteFile (lpszEncodedFile);
      LocalFree (lpszEncodedFile);
    }

    // Go to the next attached file.
    mAtt.uiAttachmentNumber++;
  }   

  return TRUE;
}

The SendAttFile function checks to see if there are any attached files in the specified mail message. SendAttFile then encodes the attached files. SendAttFile also contains an area for implementing your own code to mail the encoded data to another application.

The following code example shows how to use TnefBuildInfo. 

BOOL SendTNEFile (HANDLE hMail, MailMsg mm)
{
  HANDLE hFile;                     // Handle to the TNEF file
  
  TCHAR szTnefCorrelator[200];      // The szTnefCorrelator of the TNEF
                                    // structure
  LPWSTR lpszMsgType = NULL;        // Pointer to the message type

  TNEFINFO ti;                      // Pointer to the TNEFINFO structure
  MailNonIpmInfo* lpnii;            // Describes non-IPM data
  
  // Get mail message type.
  if (!(lpszMsgType = MailGetField (&mm, _T("Type"), FALSE)))
    return FALSE;

    // Check if the message contains schedule plus data.
    if (lstrcmp(lpszMsgType, _T("IPM.Schedule.Meeting")) != 0 &&
      lstrcmp(lpszMsgType, _T("IPM.Microsoft Schedu")) != 0)
  {
    LocalFree (lpszMsgType);
    return FALSE;
  }
  
  LocalFree (lpszMsgType);
  
  if (lpnii = HeapAlloc (mm.hHeap, 0, 
              sizeof (MailNonIpmInfo) + 30 * sizeof (CEPROPVAL)))
  {
    DWORD dwIndex = 0,            
          dwBufLen = 0;         
    LPBYTE lpbBuf = NULL;       
    
    int iPropIDs = 30;          // The number of property identifiers

    CEOID CeOid,                // The object identifier of the parent
                                   database 
          CeOidMailMsg;         // Object identifier of the mail message
    CEOIDINFO OidInfo;          // Contains the mail message information
    CEPROPID rgPropId[30];      // The identifiers of the property value

    HANDLE hDataBase = INVALID_HANDLE_VALUE;
                                // The handle to the database that 
                                // contains the mail message
    // Retrieves information about the mail message in the object store 
    if (!CeOidGetInfo (mm.oid, &OidInfo)) 
      return FALSE;

    CeOid = OidInfo.infRecord.oidParent;

    // Insert code here to open the database that contains the mail 
    // message.
    // hDataBase = ...
  
    lpnii->uiPropCount = iPropIDs;
    lpnii->szMessageClass = lpszMsgType;

    // Seeks the mail message record in the database 
    if ((CeOidMailMsg = CeSeekDatabase (
                                        hDataBase, 
                                        CEDB_SEEK_CEOID, 
                                        mm.oid, 
                                        &dwIndex)) != mm.oid) 
    {
      return FALSE;
    }
    else
    {
      // Assign identifiers of the property value to rgPropId.
      // rgPropId = {...}

      // Reads properties from the current mail record
      CeOidMailMsg = CeReadRecordProps (
                                    hDataBase,
                                    CEDB_ALLOWREALLOC,
                                    &(USHORT)lpnii->uiPropCount,
                                    rgPropId,
                                    &lpbBuf,
                                    &dwBufLen);
    }

    // Close the database.
    CloseHandle (hDataBase);
  
    // Copy propvals into lpnii.
    memcpy (&(lpnii->propArray), lpbBuf, iPropIDs * sizeof (CEPROPVAL));

    LocalFree (lpbBuf);
  }
  else
    return FALSE;
    

  // Constructing szTnefCorrelator by using an object identifier
  wsprintf (szTnefCorrelator, _T("WCETnefCorrelator%lx"), mm.oid);

  if (GetTempPath (MAX_PATH, (LPWSTR)&ti.szTnefFileName) == 0)
    lstrcpy((LPWSTR)&ti.szTnefFileName, _T("\\"));

  lstrcat ((LPWSTR)&ti.szTnefFileName, _T("WCEMail.Tnef"));

  // Fill the TNEFINFO structure.
  ti.szTnefCorrelator = szTnefCorrelator;
  ti.wContentEncoding = CODE_B64;

  // Create the encoded TNEF file.
  if (!TnefBuildInfo (&ti, MailGetField (&mm, _T("Subject"), FALSE), 
                     lpnii, NULL))
  {
    return FALSE;
  }
 
  // Open the TNEF file for reading.
  if ( (hFile = CreateFile (
                            (LPWSTR)&ti.szTnefFileName, 
                            GENERIC_READ, 
                            FILE_SHARE_READ,
                            (LPSECURITY_ATTRIBUTES)0, 
                            OPEN_EXISTING,
                            FILE_ATTRIBUTE_NORMAL, 
                            (HANDLE)0)) != INVALID_HANDLE_VALUE)
  {
    // Insert code here for reading data from the TNEF file, and then 
    // sending the data out.
    // ...

    // Once reading and sending the data are done, close the file.
    CloseHandle(hFile);
  }

  // Delete the TNEF file.
  DeleteFile ((LPWSTR) &ti.szTnefFileName);

  return TRUE;
}

The SendTNEFile function checks whether a mail message contains non-IPM data from Schedule+. If the mail message does contain this information, SendTNEFile encodes the mail message and sends out the encoded file.
