PNData_t

Enumerates the types of data used by methods such as GetData and SetData.

Syntax

typedef enum ePNData_t{
PN_DT_ALL = MIN_PNDATA_T,
PN_DT_POSITION = 1,
PN_DT_VELOCITY = 2,
PN_DT_DEVICESTATE = 3,
PN_DT_TIME = 4,
PN_DT_TM = 5,
PN_DT_ACCURACY = 6,
PN_DT_STATION = 7,
PN_DT_DEVICE = 8,
PN_DT_CONFIG = 9,
PN_DT_SETTINGS = 10,
PN_DT_ALMANAC = 14,
}PNData_t;
