Using the Contacts Database

The following example attempts to open the Contacts database by calling the OpenAddressBook function. If the function fails because the database does not exist, the example creates the database by calling the CreateAddressBook function. If the creation is successful, the example tries to open the database again. The example uses the GetPropertyDataStruct function to examine address card properties to ensure that the database has not been corrupted. The example returns TRUE if the database was successfully opened, or FALSE otherwise:

// OpenUpTheAddressBook - Opens the Contacts database, creating it

// if it doesn't exist. Returns TRUE if successful or FALSE otherwise.

// hwndParent - handle of the parent window that is to receive database

// notifications (DB_PEGOID_*).

// hhSortProp - property tag for setting the sort order.

BOOL OpenUpTheAddressBook(HWND hwndParent, HHPRTAG hhSortProp)

{-

DWORD dwError; // Error code

HHPRTAG propList[MAX_COLUMNS]; // Sort-order property tags

int nColumns; // Number of sortable columns

int index; // Loop index

// Attempt to open the Contacts database.

if (!OpenAddressBook(hwndParent, hhSortProp)) {-

dwError = GetLastError();

if (dwError == ERROR_FILE_NOT_FOUND) {-

// The Contacts database doesn't exist, so try to create

// it. Use the default sort order properties.

if (!CreateAddressBook(NULL, 0)) {-

goto InitError;

}

// The Contacts database was created, now try

// again to open it.

if (!OpenAddressBook(hwndParent, hhSortProp)) {-

goto InitError;

}

// Check for an invalid parameter error.

} else if (dwError == ERROR_INVALID_PARAMETER) {-

// Another application may have changed the sort orders.

// Try opening the database without specifying a

// sort order.

if (!OpenAddressBook(hwndParent, 0)) {-

goto InitError;

}

// Retrieve the column properties.

nColumns = MAX_COLUMNS;

if (!GetColumnProperties(&propList, &nColumns)) {-

goto InitError;

}

// Make sure the properties are valid.

for (index = 0; index < nColumns; index++) {-

if (GetPropertyDataStruct(GPDS_PROPERTY,

propList[index], NULL) == GPDS_ERR) {-

// The database is corrupt.

MessageBox(hwndParent, IDS_CORRUPT_DATABASE,

MB_ICONEXCLAMATION | MB_OK);

goto InitError;

}

}

// Try opening the database one last time.

hhSortProp = propList[0];

if (!OpenAddressBook(hwndParent, hhSortProp)) {-

goto InitError;

}

} else {-

goto InitError;

}

}

return TRUE;

InitError:

return FALSE;

}
