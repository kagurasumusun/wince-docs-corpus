Using Command Bars

The following function creates a command bar that contains six buttons, two separators, a menu bar, a combo box, and a close button. All six buttons include images provided by the system-defined button images, and three of the buttons also include strings - cut, copy, and paste.

// DoCommandBar - creates a command bar and adds items to it.

// Returns the handle of the command bar.

// hwndParent - handle of the main window

//

// Global variables:

// g_hInst - handle of the application instance

// g_hMenu - handle of the menu bar

HWND DoCommandBar(HWND hwndParent)

{-

// Button definitions.

static TBBUTTON tbButton[ ] = {-

{-0, 0, TBSTATE_ENABLED, TBSTYLE_SEP,

0, 0, 0, -1},

{-STD_FILENEW, IDC_NONE, TBSTATE_ENABLED, TBSTYLE_BUTTON,

0, 0, 0, -1},

{-STD_FILEOPEN, IDC_NONE, TBSTATE_ENABLED, TBSTYLE_BUTTON,

0, 0, 0, -1},

{-STD_FILESAVE, IDC_NONE, TBSTATE_ENABLED, TBSTYLE_BUTTON,

0, 0, 0, -1},

{-0, 0, TBSTATE_ENABLED, TBSTYLE_SEP,

0, 0, 0, -1},

{-STD_CUT, IDC_NONE, TBSTATE_ENABLED, TBSTYLE_BUTTON,

0, 0, 0, 0},

{-STD_COPY, IDC_NONE, TBSTATE_ENABLED, TBSTYLE_BUTTON,

0, 0, 0, 1},

{-STD_PASTE, IDC_NONE, TBSTATE_ENABLED, TBSTYLE_BUTTON,

0, 0, 0, 2},

};

// Create the command bar.

hwndCB = CommandBar_Create(g_hInst, hwndParent, ID_COMMANDBAR);

// Insert the menu bar, retrieve the menu handle, and save the

// handle in a global variable.

CommandBar_InsertMenubar(hwndCB, g_hInst, IDM_MAINMENU, 0);
g_hMenu = CommandBar_GetMenu(hwndCB, 0);

// Add "Cut", "Copy", and "Paste" to the list of command-bar

// strings.

SendMessage(hwndCB, TB_ADDSTRING, (WPARAM) g_hInst,
MAKELONG(IDS_CUT, 0));
SendMessage(hwndCB, TB_ADDSTRING, (WPARAM) g_hInst,
MAKELONG(IDS_COPY, 0));
SendMessage(hwndCB, TB_ADDSTRING, (WPARAM) g_hInst,
MAKELONG(IDS_PASTE, 0));

// Add the bitmap containing the system-defined button images to the

// list of command-bar button images.

CommandBar_AddBitmap(hwndCB, HINST_COMMCTRL, IDB_STD_SMALL_COLOR,

NUM_IMAGES, 0, 0);

// Add the buttons.

CommandBar_AddButtons(hwndCB, sizeof(tbButton)/sizeof(TBBUTTON),

tbButton);

// Insert the combo box.

hwndCombo = CommandBar_InsertComboBox(hwndCB, g_hInst, COMBO_WIDTH,

CBS_DROPDOWNLIST | WS_VSCROLL, ID_COMBOBOX, 9);

// Call an application-defined function to add strings to the

// combo box.

DoFillComboBox(hwndCombo);

// Add the close button.

CommandBar_AddAdornments(hwndCB, 0, IDM_EXIT);

return hwndCB;

}
