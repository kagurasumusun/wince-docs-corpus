Using a PC Card Serial Device

Using a PC Card serial device is also similar to the steps for using the built-in serial device. Note that for the PC Card device, EscapeCommFunction is not called to redirect communication. Rather, if the PC Card is a standard serial device, you must call CreateFile using “COM2:” as the lpFileName parameter.

The following sample code fragment demonstrates an example of how a PC Card serial device can be opened and setup.

//-------------------------------------------------------------------

int WINAPI  WinMain(HINSTANCE hInst, HINSTANCE hInstPrev, LPWSTR

pszCmdLine, int nCmdShow)

{-

HANDLE hPort;

DCB  PortDCB;

// Open COM2.

hPort = CreateFile(TEXT("COM2:"), GENERIC_READ | GENERIC_WRITE,

0, NULL, OPEN_EXISTING, 0, NULL);

if (hPort == INVALID_HANDLE_VALUE) {-

DbgPrintfW(TEXT("Unable to open COM2 port.\r\n"));

return FALSE;

}

// Set state of COM2 port.

if (!GetCommState(hPort, &PortDCB)) {-

DbgPrintfW((TEXT("GetCommState() error=%d.\r\n")), GetLastError());

return FALSE;

}

PortDCB.BaudRate = 9600;

PortDCB.fBinary = TRUE;

PortDCB.fParity = TRUE;

PortDCB.fOutxCtsFlow = FALSE; // ignore possible hangups

PortDCB.fOutxDsrFlow = FALSE; // don't wait on the DSR line

PortDCB.fDtrControl = DTR_CONTROL_ENABLE;

PortDCB.fDsrSensitivity = FALSE;

PortDCB.fTXContinueOnXoff = TRUE;

PortDCB.fOutX = FALSE; // no XON/XOFF control

PortDCB.fInX = FALSE;

PortDCB.fErrorChar = FALSE;

PortDCB.fNull = FALSE;

PortDCB.fRtsControl = RTS_CONTROL_ENABLE;

PortDCB.fAbortOnError = FALSE;

PortDCB.ByteSize = 8;

PortDCB.Parity = ODDPARITY;

PortDCB.StopBits = ONESTOPBIT;

if (!SetCommState(hPort, &PortDCB)) {-

DbgPrintfW((TEXT("SetCommState() error=%d.\r\n")),
GetLastError());

return FALSE;

}

SetCommMask(hPort, EV_RXCHAR);

// Read data from the PC card

getDataPackets(hPort);

// Write data to the PC card

putDataPackets(hPort);

// Close the port and exit.

SetCommMask(hPort, 0);
CloseHandle(hPort);

return TRUE;

}
