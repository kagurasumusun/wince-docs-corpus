NodeDataUnion

This union holds the raw data for an SDP element.

typedef union NodeDataUnion {
  DP_LARGE_INTEGER_16 int128;
  SDP_ULARGE_INTEGER_16 uint128;
  GUID uuid128;
  ULONG uuid32;
  USHORT uuid16;
  LONGLONG int64;
  ULONGLONG uint64;
  LONG int32;
  ULONG uint32;
  SHORT int16;
  USHORT uint16;
  CHAR int8;
  UCHAR uint8;
  UCHAR booleanVal;
  SdpString str;
  SdpString url;
  ISdpNodeContainer* container;
} NodeDataUnion;
